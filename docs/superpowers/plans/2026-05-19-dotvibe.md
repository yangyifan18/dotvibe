> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Go CLI tool that backs up and restores vibe coding agent data (Claude Code, Codex CLI, OpenCode) across Mac machines.

**Architecture:** Single-binary Go CLI with a pluggable adapter pattern. Each tool (Claude Code, Codex CLI, OpenCode) has an adapter that knows how to detect, list, and restore its data. The `backup` package handles tar.gz archive creation/reading and manifest generation. The `config` package handles exclusion rules.

**Tech Stack:** Go 1.22+, cobra (CLI), promptui (interactive selection), standard library archive/tar + compress/gzip

```go
// Category constants for file classification.
const (
	CategoryConfig  = "config"
	CategoryMemory  = "memory"

// FileEntry represents a single file to back up or restore.
type FileEntry struct {
	SourcePath string // Absolute path on disk
	InArchive  string // Path inside the tar.gz archive
	Category   string // config, memory, skills, history
	Size       int64
}

// ExportOpts controls what gets included in a backup.
type ExportOpts struct {
	WithHistory     bool
	ExcludePatterns []string
	OnlyTools       []string
}

// ToolStatus holds detected info about an installed tool.
type ToolStatus struct {
	Name       string
	Path       string
	Size       int64
	Projects   int
	Skills     int
	Agents     int
	Sessions   int
	ConfigFile string
}

// RestoreOpts controls how restore behaves.
type RestoreOpts struct {
	Force   bool
	Project string // restore specific project only
}

// Adapter is the interface each supported tool must implement.
type Adapter interface {
	// Name returns the tool's display name (e.g. "Claude Code").
	Name() string

	// ID returns the tool's slug (e.g. "claude-code").
	ID() string

	// Detect returns true if the tool is installed.
	Detect() bool

	// ListFiles returns all files that should be backed up.
	ListFiles(opts ExportOpts) []FileEntry

	// Status returns a summary of the tool's local data.
	Status() ToolStatus

	// RestoreFiles restores the given entries from the archive.
	RestoreFiles(entries []FileEntry, archiveDir string, opts RestoreOpts) error
}

// AllAdapters returns all supported tool adapters.
func AllAdapters() []Adapter {
	return []Adapter{
		NewClaudeAdapter(),
		NewCodexAdapter(),
		NewOpenCodeAdapter(),
	}
```

```go
package config

import (
	"path/filepath"
	"strings"
)

// defaultExcludeDirs are always excluded regardless of user options.
var defaultExcludeDirs = []string{
	"auth.json",
	"telemetry",
	"cache",
	"session-env",
	"shell-snapshots",
}

// Excluder checks whether a given archive path should be excluded.
type Excluder struct {
	customPatterns []string
}

// NewExcluder creates an Excluder with optional user-defined glob patterns.
func NewExcluder(customPatterns []string) *Excluder {
	return &Excluder{customPatterns: customPatterns}
}

// IsExcluded returns true if the path matches any exclusion rule.
// path is the relative path inside the archive (e.g. "claude-code/telemetry/event.json").
func (e *Excluder) IsExcluded(path string) bool {
	// Check default exclusions by path segments
	segments := strings.Split(filepath.ToSlash(path), "/")
	for _, seg := range segments {
		for _, exc := range defaultExcludeDirs {
			if seg == exc {
				return true
			}
		}
	}

	// Check custom glob patterns
	for _, pattern := range e.customPatterns {
		if matchGlob(pattern, path) {
			return true
		}
	}

	return false
}

// matchGlob checks if path matches a glob pattern.
// Supports ** for directory matching and * for filename matching.
func matchGlob(pattern, path string) bool {
	// Use filepath.Match for simple patterns
	matched, err := filepath.Match(pattern, path)
	if err == nil && matched {
		return true
	}

	// Also try matching against just the base name for patterns like "transcripts/*.jsonl"
	base := filepath.Base(path)
	matched, err = filepath.Match(pattern, base)
	if err == nil && matched {
		return true
	}

	// Try matching pattern against each suffix of the path
	parts := strings.Split(filepath.ToSlash(path), "/")
	for i := range parts {
		suffix := strings.Join(parts[i:], "/")
		matched, err = filepath.Match(pattern, suffix)
		if err == nil && matched {
			return true
		}
	}

	return false
}
```

```go
package backup

import (
	"encoding/json"
	"os"
	"time"
)

// Manifest is the metadata stored in every dotvibe backup.
type Manifest struct {
	Version  string                  `json:"version"`
	Created  time.Time               `json:"created"`
	Hostname string                  `json:"hostname"`
	Tools    map[string]ToolManifest `json:"tools"`
}

// ToolManifest describes one tool's data in the backup.
type ToolManifest struct {
	Version      string   `json:"version,omitempty"`
	Included     []string `json:"included"`
	ProjectCount int      `json:"project_count,omitempty"`
	FileCount    int      `json:"file_count,omitempty"`
	AgentCount   int      `json:"agent_count,omitempty"`
}

// WriteManifest writes a manifest to disk as pretty JSON.
func WriteManifest(path string, m *Manifest) error {
	data, err := json.MarshalIndent(m, "", "  ")
	if err != nil {
		return err
	}
	return os.WriteFile(path, data, 0644)
}

// ReadManifest reads a manifest from disk.
func ReadManifest(path string) (*Manifest, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	var m Manifest
	if err := json.Unmarshal(data, &m); err != nil {
		return nil, err
	}
	return &m, nil
}
```

```go
package backup

import (
	"archive/tar"
	"compress/gzip"
	"io"
	"os"

	"github.com/young/dotvibe/adapters"
)

// CreateArchive writes a tar.gz file containing manifest and all entries.
func CreateArchive(dst string, manifest *Manifest, entries []adapters.FileEntry) error {
	f, err := os.Create(dst)
	if err != nil {
		return err
	}
	defer f.Close()

	gw := gzip.NewWriter(f)
	defer gw.Close()

	tw := tar.NewWriter(gw)
	defer tw.Close()

	// Write manifest first
	if err := writeManifestToTar(tw, manifest); err != nil {
		return err
	}

	// Write all file entries
	for _, entry := range entries {
		if err := writeFileToTar(tw, entry.SourcePath, entry.InArchive); err != nil {
			return err
		}
	}

	return nil
```

```go
// ReadArchive opens a tar.gz for reading. Caller must call Close().
type ArchiveReader struct {
	f        *os.File
	gz       *gzip.Reader
	Manifest *Manifest
	files    []archiveFile
}

type archiveFile struct {
	Name string
	Size int64
	Data []byte // lazily loaded
}

// ReadArchive opens and indexes an archive.
func ReadArchive(path string) (*ArchiveReader, error) {
	f, err := os.Open(path)
	if err != nil {
		return nil, err
	}

	gz, err := gzip.NewReader(f)
	if err != nil {
		f.Close()
		return nil, err
	}

	ar := &ArchiveReader{f: f, gz: gz}
	if err := ar.index(); err != nil {
		f.Close()
		return nil, err
	}

	return ar, nil
}

func (ar *ArchiveReader) index() error {
	tr := tar.NewReader(ar.gz)
	for {
		hdr, err := tr.Next()
		if err == io.EOF {
			break
		}
		if err != nil {
			return err
		}

		if hdr.Name == "manifest.json" {
			data, err := io.ReadAll(tr)
			if err != nil {
				return err
			}
			var m Manifest
			if err := json.Unmarshal(data, &m); err != nil {
				return err
			}
			ar.Manifest = &m
			ar.files = append(ar.files, archiveFile{Name: hdr.Name, Size: hdr.Size, Data: data})
		} else {
			data, err := io.ReadAll(tr)
			if err != nil {
				return err
			}
			ar.files = append(ar.files, archiveFile{Name: hdr.Name, Size: hdr.Size, Data: data})
		}
	}
	return nil
}

// ListFiles returns the paths of all files in the archive (excluding manifest).
func (ar *ArchiveReader) ListFiles() []string {
	var names []string
	for _, f := range ar.files {
		if f.Name != "manifest.json" {
			names = append(names, f.Name)
		}
	}
	return names
}

// ReadFile returns the contents of a file from the archive.
func (ar *ArchiveReader) ReadFile(name string) ([]byte, error) {
	for _, f := range ar.files {
		if f.Name == name {
			return f.Data, nil
		}
	}
	return nil, fmt.Errorf("file not found in archive: %s", name)
}

// Close releases all resources.
func (ar *ArchiveReader) Close() error {
	ar.gz.Close()
	return ar.f.Close()
}
```

```go
func (a *ClaudeAdapter) adaptPath(archivePath string) string {

	// "claude-code/config/settings.json" → ~/.claude/settings.json

	parts := strings.SplitN(rel, "/", 2)
```

```go
func formatSize(bytes int64) string {
	const (
		KB = 1024
		MB = 1024 * KB
		GB = 1024 * MB
	)

	switch {
	case bytes >= GB:
		return fmt.Sprintf("%.1f GB", float64(bytes)/float64(GB))
	case bytes >= MB:
		return fmt.Sprintf("%.0f MB", float64(bytes)/float64(MB))
	case bytes >= KB:
		return fmt.Sprintf("%.0f KB", float64(bytes)/float64(KB))
	default:
		return fmt.Sprintf("%d B", bytes)
	}
}
```

```go
			files := adapter.ListFiles(opts)

			// Apply exclusions

			for _, f := range files {
				if !excluder.IsExcluded(f.InArchive) {
					filtered = append(filtered, f)
				}
			}

			fmt.Printf("%d files\n", len(filtered))
			allEntries = append(allEntries, filtered...)

			// Build tool manifest
			categories := map[string]bool{}
			for _, f := range filtered {
				categories[f.Category] = true
			}
			var included []string
			for cat := range categories {
				included = append(included, cat)
			}

			toolManifests[adapter.ID()] = backup.ToolManifest{
				Included:  included,
				FileCount: len(filtered),
			}
		}

		// Create manifest
		hostname, _ := os.Hostname()
		manifest := &backup.Manifest{
			Version:  "1.0.0",
			Created:  time.Now().UTC(),
			Hostname: hostname,
			Tools:    toolManifests,
		}

		if err := backup.CreateArchive(output, manifest, allEntries); err != nil {
			return fmt.Errorf("failed to create archive: %w", err)
		}

func splitAndTrim(s string) []string {
	var result []string
	for _, part := range splitString(s, ",") {
		trimmed := trimSpace(part)
		if trimmed != "" {
			result = append(result, trimmed)
		}
	}
	return result
}

func splitString(s, sep string) []string {
	if s == "" {
		return nil
	}
	result := []string{}
	start := 0
	for i := 0; i <= len(s)-len(sep); i++ {
		if s[i:i+len(sep)] == sep {
			result = append(result, s[start:i])
			start = i + len(sep)
		}
	}
	result = append(result, s[start:])
	return result
}

func trimSpace(s string) string {
	start, end := 0, len(s)
	for start < end && (s[start] == ' ' || s[start] == '\t') {
		start++
	}
	for end > start && (s[end-1] == ' ' || s[end-1] == '\t') {
		end--
	}
	return s[start:end]
}

func containsTool(tools []string, id string) bool {
	for _, t := range tools {
		if t == id {
			return true
		}
	}
	return false
}
```

```go
// ExtractArchive extracts all files from a tar.gz to a directory.
func ExtractArchive(archivePath, destDir string) error {
	f, err := os.Open(archivePath)
	if err != nil {
		return err
	}
	defer f.Close()

	gz, err := gzip.NewReader(f)
	if err != nil {
		return err
	}
	defer gz.Close()

	tr := tar.NewReader(gz)
	for {
		hdr, err := tr.Next()
		if err == io.EOF {
			break
		}
		if err != nil {
			return err
		}

		target := filepath.Join(destDir, hdr.Name)

		if err := os.MkdirAll(filepath.Dir(target), 0755); err != nil {
			return err
		}

		f, err := os.Create(target)
		if err != nil {
			return err
		}

		if _, err := io.Copy(f, tr); err != nil {
			f.Close()
			return err
		}
		f.Close()
	}

	return nil
}
```
